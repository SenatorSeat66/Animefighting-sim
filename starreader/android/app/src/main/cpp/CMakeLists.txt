cmake_minimum_required(VERSION 3.22)
project(starreader_quickjs LANGUAGES C CXX)
set(CMAKE_C_STANDARD 11)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(ENGINE "${CMAKE_CURRENT_LIST_DIR}/../../../../../third_party/quickjs")
file(GLOB BINDING CONFIGURE_DEPENDS "binding/*.cpp")
add_library(quickjs SHARED ${BINDING}
  "${ENGINE}/quickjs.c" "${ENGINE}/dtoa.c" "${ENGINE}/cutils.c"
  "${ENGINE}/libregexp.c" "${ENGINE}/libunicode.c")
target_include_directories(quickjs PRIVATE "${ENGINE}/..")
target_compile_definitions(quickjs PRIVATE _GNU_SOURCE CONFIG_VERSION="2026-06-04")
target_compile_options(quickjs PRIVATE -fvisibility=hidden)
target_link_libraries(quickjs PRIVATE m)
if(ANDROID)
  target_link_options(quickjs PRIVATE "-Wl,-z,max-page-size=16384")
else()
  find_package(JNI REQUIRED)
  target_include_directories(quickjs PRIVATE ${JNI_INCLUDE_DIRS})
endif()
